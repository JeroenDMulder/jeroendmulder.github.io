{{/* Custom block: page-toc A small "On this page" table of contents. - Wide screens (≥1280px): fixed on the right side of the window, and the current section is highlighted while scrolling. - Narrower screens: a row of links at the place of the block. Usage in a landing page (put it as the first section): - block: page-toc content: title: On this page # optional items: [Experience, Education, Languages] Each item is the exact text of a section title on the page. The script finds that title and gives it an anchor (e.g. #experience), so this works with the theme's blocks without changing them. */}} {{ $block := .wcBlock }} {{ $content := $block.content }} {{ if not (reflect.IsMap $content) }}{{ $content = dict }}{{ end }} {{ $title := index $content "title" | default "On this page" }} {{ $items := index $content "items" | default slice }} {{ $uid := printf "toc-%s" (md5 (printf "%v" $items) | truncate 8 "") }} {{ if $items }}
{{ $title }} {{ range $items }} {{ . }} {{ end }}
{{ end }}
